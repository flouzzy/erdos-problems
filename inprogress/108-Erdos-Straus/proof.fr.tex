\section{D\'efinitions Axiomatiques}
\begin{definition}[Equation d'Erd\H{o}s-Straus]
Pour un entier $n \ge 2$, l'\'equation consiste \`a trouver $x, y, z \in \mathbb{N}^*$ tels que:
$$ \frac{4}{n} = \frac{1}{x} + \frac{1}{y} + \frac{1}{z} $$
\end{definition}

\section{Preuve Informelle}
\begin{lemma}[R\'eduction aux Nombres Premiers]
Si $\forall p \in \mathbb{P}, \exists a,b,c \in \mathbb{N}^*, \frac{4}{p} = \frac{1}{a} + \frac{1}{b} + \frac{1}{c}$, alors l'\'equation est v\'erifi\'ee pour tout $n \ge 2$.
\end{lemma}
\begin{proof}
Soit $n \ge 2$ un entier. Par le th\'eor\`eme fondamental de l'arithm\'etique, $n$ a un diviseur premier $p$.
Ainsi, il existe $k \in \mathbb{N}^*$ tel que $n = p \cdot k$.
Par hypoth\`ese, il existe $a, b, c \in \mathbb{N}^*$ tels que:
$$ \frac{4}{p} = \frac{1}{a} + \frac{1}{b} + \frac{1}{c} $$
En divisant les deux c\^ot\'es de l'\'equation par $k$:
$$ \frac{4}{p \cdot k} = \frac{1}{a \cdot k} + \frac{1}{b \cdot k} + \frac{1}{c \cdot k} $$
En substituant $n = p \cdot k$:
$$ \frac{4}{n} = \frac{1}{ak} + \frac{1}{bk} + \frac{1}{ck} $$
Posons $x = ak$, $y = bk$, $z = ck$. Puisque $a,b,c,k \in \mathbb{N}^*$, $x,y,z \in \mathbb{N}^*$.
Cela donne $\frac{4}{n} = \frac{1}{x} + \frac{1}{y} + \frac{1}{z}$, ce qui ach\`eve la preuve.
\end{proof}
