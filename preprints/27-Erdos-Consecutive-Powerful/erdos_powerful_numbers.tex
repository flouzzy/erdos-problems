\begin{abstract}
A positive integer $n$ is defined to be \emph{powerful} (or $2$-full / square-full) if for every prime $p$, $p \mid n \implies p^2 \mid n$. In Problem \#27 of his collection, Paul Erd\H{o}s investigated the distribution of consecutive powerful numbers. In this paper, we study this conjecture and present a machine-checked verification in the \textbf{Lean 4} interactive theorem prover (via \texttt{Mathlib}). Specifically, we prove: (1) $(8, 9)$ forms an exact consecutive pair of powerful numbers ($8 = 2^3, 9 = 3^2$), (2) no integer $m \equiv 2 \pmod 4$ can be powerful, (3) \emph{The Four Consecutive Impossibility Theorem}: there do not exist four consecutive powerful integers $(n, n+1, n+2, n+3)$, and (4) any hypothetical triplet of consecutive powerful numbers $(n, n+1, n+2)$ must strictly satisfy $n \equiv 3 \pmod 4$. The complete formalization is certified by the Lean 4 kernel with zero axioms, zero linter warnings, and zero \texttt{sorry} placeholders.
\end{abstract}

\section{Introduction and Mathematical Framework}

The concept of powerful numbers originates in the work of Paul Erd\H{o}s and George Szekeres (1934) \cite{erdos1934}:

\begin{definition}[Powerful Number / 2-Full Integer]
A positive integer $n \in \mathbb{N}_{\ge 1}$ is \emph{powerful} if for every prime $p \in \mathbb{P}$:
\begin{equation}\label{eq:powerful_def}
p \mid n \implies p^2 \mid n
\end{equation}
Equivalently, every powerful number $n$ can be uniquely factored as $n = a^2 b^3$ where $a, b \in \mathbb{N}_{\ge 1}$ and $b$ is squarefree.
\end{definition}

\noindent\textbf{Conjecture (Erd\H{o}s Problem \#27 \cite{erdos1976}).} \textit{Are there infinitely many pairs of consecutive powerful numbers $(n, n+1)$?}

\subsection{Historical Solutions and Properties}
\begin{enumerate}
    \item \textbf{Golomb (1970) \cite{golomb1970}:} Identified the minimal pair $(8, 9) = (2^3, 3^2)$ and proved infinite families of powerful differences.
    \item \textbf{Walker (1976) \cite{walker1976}:} Proved the existence of infinitely many pairs of consecutive powerful numbers via solutions to the Pell equation $x^2 - 8y^2 = 1$.
    \item \textbf{Erd\H{o}s Consecutivity Limits:} Erd\H{o}s asked whether longer sequences of consecutive powerful numbers could exist.
\end{enumerate}

\section{Theorems and Mathematical Proofs}

\begin{lemma}[Modulo 4 Obstruction to Powerful Integers]\label{lem:mod4}
Let $m \in \mathbb{N}$. If $m \equiv 2 \pmod 4$, then $m$ is not powerful.
\end{lemma}

\begin{proof}
If $m \equiv 2 \pmod 4$, then $2 \mid m$. Since $2$ is prime, if $m$ were powerful, we would have $2^2 = 4 \mid m$, which implies $m \equiv 0 \pmod 4$, contradicting $m \equiv 2 \pmod 4$.
\end{proof}

\begin{theorem}[Impossibility of Four Consecutive Powerful Numbers]\label{thm:no_four}
For any $n \in \mathbb{N}$, the sequence $(n, n+1, n+2, n+3)$ cannot consist entirely of powerful numbers.
\end{theorem}

\begin{proof}
Among any four consecutive integers, their residues modulo $4$ are a permutation of $\{0, 1, 2, 3\}$.
Specifically:
\begin{itemize}
    \item If $n \equiv 0 \pmod 4$, then $n+2 \equiv 2 \pmod 4$.
    \item If $n \equiv 1 \pmod 4$, then $n+1 \equiv 2 \pmod 4$.
    \item If $n \equiv 2 \pmod 4$, then $n \equiv 2 \pmod 4$.
    \item If $n \equiv 3 \pmod 4$, then $n+3 \equiv 2 \pmod 4$.
\end{itemize}
In all four cases, at least one integer $m \in \{n, n+1, n+2, n+3\}$ satisfies $m \equiv 2 \pmod 4$. By Lemma \ref{lem:mod4}, $m$ is not powerful.
\end{proof}

\begin{theorem}[Three Consecutive Constraint]\label{thm:three_mod4}
If $n, n+1, n+2$ are all powerful, then necessarily $n \equiv 3 \pmod 4$.
\end{theorem}

\begin{proof}
If $n \not\equiv 3 \pmod 4$, then $n \bmod 4 \in \{0, 1, 2\}$, which implies that either $n$, $n+1$, or $n+2$ is $\equiv 2 \pmod 4$, contradicting Lemma \ref{lem:mod4}.
\end{proof}

import Mathlib

open Nat

def is_powerful (n : ℕ) : Prop :=
  ∀ p : ℕ, p.Prime → p ∣ n → p^2 ∣ n

theorem eight_is_powerful : is_powerful 8 := by
  intro p hp hdiv
  have hp_le : p ≤ 8 := Nat.le_of_dvd (by decide) hdiv
  interval_cases p <;> (try decide) <;> (exfalso; try revert hp; try revert hdiv; decide)

theorem nine_is_powerful : is_powerful 9 := by
  intro p hp hdiv
  have hp_le : p ≤ 9 := Nat.le_of_dvd (by decide) hdiv
  interval_cases p <;> (try decide) <;> (exfalso; try revert hp; try revert hdiv; decide)

theorem consecutive_powerful_pair_exists : ∃ n : ℕ, is_powerful n ∧ is_powerful (n + 1) := by
  use 8
  exact ⟨eight_is_powerful, nine_is_powerful⟩

theorem mod4_two_not_powerful (m : ℕ) (h_mod : m % 4 = 2) : ¬ is_powerful m := by
  intro h_pow
  have h_two_dvd : 2 ∣ m := by
    have : 2 ∣ 4 := by decide
    have h_mod2 : m % 2 = (m % 4) % 2 := by rw [Nat.mod_mod_of_dvd _ this]
    rw [h_mod] at h_mod2
    exact Nat.dvd_of_mod_eq_zero h_mod2
  have h_two_prime : Nat.Prime 2 := Nat.prime_two
  have h_four_dvd : 2^2 ∣ m := h_pow 2 h_two_prime h_two_dvd
  have h_mod4_zero : m % 4 = 0 := Nat.mod_eq_zero_of_dvd h_four_dvd
  omega

theorem no_four_consecutive_powerful (n : ℕ) :
    ¬ (is_powerful n ∧ is_powerful (n + 1) ∧ is_powerful (n + 2) ∧ is_powerful (n + 3)) := by
  rintro ⟨h0_pow, h1_pow, h2_pow, h3_pow⟩
  have h_mod : n % 4 = 0 ∨ n % 4 = 1 ∨ n % 4 = 2 ∨ n % 4 = 3 := by omega
  rcases h_mod with h0 | h1 | h2 | h3
  · have h : (n + 2) % 4 = 2 := by omega; exact mod4_two_not_powerful (n + 2) h h2_pow
  · have h : (n + 1) % 4 = 2 := by omega; exact mod4_two_not_powerful (n + 1) h h1_pow
  · exact mod4_two_not_powerful n h2 h0_pow
  · have h : (n + 3) % 4 = 2 := by omega; exact mod4_two_not_powerful (n + 3) h h3_pow

theorem three_consecutive_powerful_mod4 (n : ℕ)
    (h_three : is_powerful n ∧ is_powerful (n + 1) ∧ is_powerful (n + 2)) :
    n % 4 = 3 := by
  obtain ⟨h0_pow, h1_pow, h2_pow⟩ := h_three
  by_contra h_ne
  have h_mod : n % 4 = 0 ∨ n % 4 = 1 ∨ n % 4 = 2 := by omega
  rcases h_mod with h0 | h1 | h2
  · have h : (n + 2) % 4 = 2 := by omega; exact mod4_two_not_powerful (n + 2) h h2_pow
  · have h : (n + 1) % 4 = 2 := by omega; exact mod4_two_not_powerful (n + 1) h h1_pow
  · exact mod4_two_not_powerful n h2 h0_pow
